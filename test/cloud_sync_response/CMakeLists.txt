cmake_minimum_required(VERSION 3.16)
project(cloud_sync_response_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(ARDUINOJSON_INCLUDE_DIR "${REPO_ROOT}/.pio/libdeps/default/ArduinoJson/src" CACHE PATH "ArduinoJson headers")
add_executable(cloud_sync_response_test cloud_sync_response_test.cpp)
target_include_directories(cloud_sync_response_test PRIVATE "${REPO_ROOT}" "${ARDUINOJSON_INCLUDE_DIR}")
target_link_libraries(cloud_sync_response_test PRIVATE GTest::gtest_main)
target_compile_options(cloud_sync_response_test PRIVATE -Wall -Wextra -pedantic)
enable_testing()
include(GoogleTest)
gtest_discover_tests(cloud_sync_response_test)
